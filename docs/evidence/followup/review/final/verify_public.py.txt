"""Verify compact evidence and archived source without rewriting the repository."""
import hashlib,json,re,tarfile
from pathlib import Path
REPO=Path('/Users/neeraj/src/zomglings/afloat')
ROOT=REPO/'docs/evidence/followup'
OUT=Path(__file__).parent

def digest(data): return hashlib.sha256(data).hexdigest()
manifest=json.loads((ROOT/'evidence-sha256.json').read_text())
failed=[name for name,value in manifest.items() if digest((ROOT/name).read_bytes())!=value]
unlisted=[str(p.relative_to(ROOT)) for p in ROOT.rglob('*') if p.is_file() and p.name!='evidence-sha256.json' and str(p.relative_to(ROOT)) not in manifest]
archive_counts={}
for archive in ROOT.rglob('source.tar.gz'):
 envpath=archive.parent/'environment.json'
 if archive.parent.name=='accuracy':envpath=archive.parent/'gain-1-seed-10/environment.json'
 env=json.loads(envpath.read_text())
 with tarfile.open(archive) as stream:
  for name,expected in env['source_hashes'].items():
   member=stream.extractfile('source/'+name)
   assert member is not None
   assert digest(member.read())==expected, (archive,name)
 archive_counts[str(archive.relative_to(ROOT))]=len(env['source_hashes'])
for p in (ROOT/'performance').iterdir():
 if p.is_file(): assert p.read_bytes()==(REPO/'runs/followup.KGkx3v/performance'/p.name).read_bytes(),p
assert (ROOT/'analysis/analysis-source.py').read_bytes()==(REPO/'scripts/analyze_followup.py').read_bytes()
assert (ROOT/'analysis/benchmark_followup.py').read_bytes()==(REPO/'scripts/benchmark_followup.py').read_bytes()==(ROOT/'performance/benchmark-source.py').read_bytes()
broken=[]
for p in (REPO/'followup-results.md',REPO/'docs/followup-reproduction.md',ROOT/'README.md',ROOT/'audit.md'):
 for target in re.findall(r'\]\(([^)]+)\)',p.read_text()):
  if not target.startswith(('http:','https:','#')) and not (p.parent/target.split('#')[0]).exists(): broken.append([str(p),target])
result={'public_manifest_entries':len(manifest),'hash_failures':failed,'unlisted_files':unlisted,'archive_source_files_verified':archive_counts,'broken_relative_links':broken}
(OUT/'public-verified.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
assert not failed and not broken
